⚙️ Première utilisation — Créer la table dans Supabase

Copiez ce SQL dans Supabase → SQL Editor → New query et exécutez-le une seule fois :

-- Table de suivi des clics visiteurs
CREATE TABLE IF NOT EXISTS click_events (
  id         UUID        DEFAULT gen_random_uuid() PRIMARY KEY,
  page       TEXT,
  element    TEXT,
  label      TEXT,
  href       TEXT,
  section    TEXT,
  pos_x      INTEGER,
  pos_y      INTEGER,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE click_events ENABLE ROW LEVEL SECURITY;
-- Visiteurs (anonymes) peuvent insérer
CREATE POLICY "ct_insert" ON click_events FOR INSERT  WITH CHECK (true);
-- Admins connectés peuvent lire
CREATE POLICY "ct_select" ON click_events FOR SELECT  USING (auth.role() = 'authenticated');
-- Admins connectés peuvent supprimer
CREATE POLICY "ct_delete" ON click_events FOR DELETE  USING (auth.role() = 'authenticated');
